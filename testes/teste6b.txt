program teste6b begin
    x, y, _contador = int;
    opc = char;
    x = 8;
    y = 2;
    _contador = 0;
    opc = 'S';
    if ((x >= y) && (x != 0)) then
        write("")
    else
        write("menor")
    end;
    while ((x > 0) || (y <= 1)) do
        x = x - 1;
        y = y + 1
    end;
    repeat
        _contador = _contador + x % 2;
        write(-x);
        write(!(x == y));
        write(x / y * 1.0);
        read(opc)
    until ((opc == 'N') || (opc == 'n'));
    if (x < y) then
        write("fim")
    end
end.
